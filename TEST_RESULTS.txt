  BEFORE (Broken):
    def standardize(X, mean, sd):

AFTER (Fixed):
    def standardize(X, mean, sd):
        """Comprehensive docstring added"""
        if np.any(sd == 0):
            raise ValueError(...)  # Proper error handling
        return (X - mean) / sd     # Explicit return
